// Kotlin Discord Daemon (JDA) • Live RCON & Minecraft Chat Sync
package net.banglaverse.daemon

import net.dv8tion.jda.api.JDABuilder
import net.dv8tion.jda.api.events.message.MessageReceivedEvent
import net.dv8tion.jda.api.hooks.ListenerAdapter

class MinecraftBridgeDaemon(private val rcon: RconClient) : ListenerAdapter() {
    fun start(token: String) {
        JDABuilder.createDefault(token)
            .addEventListeners(this)
            .build()
            .awaitReady()
    }

    override fun onMessageReceived(event: MessageReceivedEvent) {
        if (event.author.isBot) return
        rcon.broadcast("[Discord] ${event.author.name}: ${event.message.contentDisplay}")
    }
}
